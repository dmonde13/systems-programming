#!/bin/csh
# Script name: question10
#Script displaying a menu and performing arithmetic operations

#Verify that two arguments have been provided
if ($#argv != 2) then
    echo "Usage: $0 <integer1> <integer2>"
    echo "Please provide two integers as arguments."
    exit 1
endif

#Assign the input arguments to variables
set num1 = $argv[1]
set num2 = $argv[2]

#loop
while (1)

    #Display the menu
    echo "Add the numbers (enter a or A)"
    echo "Subtract the numbers (enter s or S)"
    echo "Multiply the numbers (enter m or M)"
    echo "Divide the numbers (enter d or D)"
    echo "Exit (select e or E)"

    #Read the user's choice
    echo "Please select an option:"
    set choice = $<

    switch ($choice)
        #If user chooses Addition
        case a:
        case A:
            @ sum = $num1 + $num2
            echo "The sum of $num1 and $num2 is: $sum"
            breaksw
        #If user chooses Subtraction
        case s:
        case S:
            @ sub = $num1 - $num2
            echo "$num1 minus $num2 is: $sub"
            breaksw
        #If user chooses Multiplication
        case m:
        case M:
            @ product = $num1 * $num2
            echo "The product of $num1 and $num2 is: $product"
            breaksw
        #If user chooses Division
        case d:
        case D:
            if ($num2 == 0) then
                echo "Error: Cannot divide by zero"
            else
                set div=`echo "scale=2; $num1 / $num2" | bc`
                echo "$num1 divided by $num2 is: $div"
            endif
            breaksw
        #If user chooses to Exit
        case e:
        case E:
            echo "Exited script."
            exit 0
            breaksw
        #If user enters invalid choice
        default:
            echo "Invalid option. Please select an option based on menu"
            continue
    endsw
    
    #Ask if user wants to use another operations
    echo "Would you like to:"
    echo "1.Perform another operation on the same integers (enter 1)"
    echo "2.Perform another operation with different integers (enter 2)"
    echo "3.Exit (enter 3)"
    set nextChoice = $<
    
    switch ($nextChoice)
        #If user sticks with the same integers
        case 1:
            continue
            breaksw
        #If user changes the integers
        case 2:
            while (1)
                echo "Please enter the first integer:"
                set num1 = $< 
                echo "Please enter the second integer:"
                set num2 = $<

                if ($num1 == "" || $num2 == "") then 
                    echo "Please enter exactly two integers."
                    continue
                endif
                if (! ($num1 + 0 >= 0 && $num2 + 0 >= 0)) then
                    echo "Invalid inputs. Please enter two new integers."
                    continue
                endif
                break
            end
            continue
            breaksw
        #If user chooses to exit the script
        case 3:
            echo "Exited script."
            exit 0
            breaksw
        #if user enters an invalid choice
        default:
            echo "Invalid choice."
            exit 1
            breaksw
    endsw
end
