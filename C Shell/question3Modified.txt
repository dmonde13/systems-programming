#!/bin/csh
# Script name:  question3Modified
# Script to return user score based on command line argument

#Ensures one argument provided
if ($#argv != 1) then
    echo "Usage: $0 <score between 0 and 100>"
    exit 1
endif

set score = $argv[1]

#Validate user input is between 0 and 100
if ("$score" !~ [0-9]*) then
    echo "Error: Input invalid. Please enter a score between 0 and 100."
    exit 1
endif

if ($score < 0 || $score > 100) then
    echo "Error: Input invalid. Please enter a score between 0 and 100."
    exit 1
endif

#Determine score based on user input
if ($score >= 90) then
    echo "You got A"
else if ($score >= 80) then
    echo "You got B"
else if ($score >= 70) then
    echo "You got C"
else if ($score >= 60) then
    echo "You got D"
else
    echo "You failed"
endif

exit 0
