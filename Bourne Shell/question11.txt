#!/bin/sh
# Script name: question11

#Ask user for an integer from 1 to 10.
echo "Please enter an integer between 1 and 10:"
read number

if [ "$number" -lt 1 ] || [ "$number" -gt 10 ]; then
    echo "Error: Please select an integer between 1 and 10"
    exit 1
fi

#Initialize varaibles for sum and product
sum=0
product=1

#Calculate factorial
i=1
while [ "$i" -le "$number" ]; do
    sum=$((sum+i))
    product=$((product * i))
    i=$((i+1))
done

#Prints sum and product
echo "Sum: $sum"
echo "Factorial: $product"
