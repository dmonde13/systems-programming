#!/bin/sh
# Script name:  question2
# Script that asks the user “Are you OK?” and gives a response based on their answer
while true; do
	echo "Are you OK? (y/n)"
        read input
        if [ "$input" = "y" ] || [ "$input" = "Y" ]; then
                echo "glad to hear it"
                break
        elif [ "$input" = "n" ] || [ "$input" = "N" ]; then
                echo "sorry that you are not feeling good"
                break
        else
            	echo "Incorrect choice. Are you OK? (y/n)"
        fi
done
exit 0
